#### Step 1：定义核心抽象 (`pipeline/base.py`)

```python
"""管线核心抽象"""
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Optional, Any, Callable, Awaitable
from enum import Enum
import time


class StepStatus(str, Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    SKIPPED = "skipped"
    FAILED = "failed"
    COMPENSATED = "compensated"


@dataclass
class StepResult:
    """步骤执行结果"""
    status: StepStatus
    data: dict = field(default_factory=dict)
    error: Optional[str] = None
    duration_ms: float = 0.0
    warnings: list[str] = field(default_factory=list)

    @property
    def ok(self) -> bool:
        return self.status == StepStatus.COMPLETED


@dataclass
class PipelineContext:
    """管线执行上下文 — 贯穿所有步骤的共享状态"""
    book_id: str
    chapter: int
    mode: str = "gacha_parallel_3"
    pipeline_id: str = ""
    
    # 步骤间共享的中间结果
    kg_snapshot_id: str = ""
    kg_summary: str = ""
    blueprint: Optional[dict] = None
    draft: str = ""
    audit_result: Optional[dict] = None
    society_insights: Optional[dict] = None
    polished_draft: str = ""
    revision_count: int = 0
    max_revisions: int = 3
    
    # 自适应标记
    chapter_type: str = "normal"
    budget_tier: str = "medium"
    skip_society: bool = False
    skip_icu: bool = False
    
    # 元数据
    start_time: float = field(default_factory=time.time)
    extra: dict = field(default_factory=dict)


class PipelineStep(ABC):
    """管线步骤基类（对标 Haystack Component + Prefect Task）
    
    子类必须实现:
    - name: 步骤名称
    - requires: 依赖的前置步骤列表
    - run(): 执行逻辑
    - rollback(): 补偿操作（可选）
    """
    name: str = ""
    requires: list[str] = []
    optional: bool = False
    max_retries: int = 1
    timeout_seconds: float = 300.0

    async def should_run(self, ctx: PipelineContext) -> bool:
        """决定是否执行此步骤（子类可覆写做条件跳过）"""
        return True

    @abstractmethod
    async def run(self, ctx: PipelineContext) -> StepResult:
        """执行步骤逻辑"""
        ...

    async def rollback(self, ctx: PipelineContext) -> None:
        """补偿操作（对标 Saga Pattern），默认无操作"""
        pass

    async def __call__(self, ctx: PipelineContext) -> StepResult:
        """框架调用入口：记录耗时 + 错误处理"""
        t0 = time.perf_counter()
        try:
            if not await self.should_run(ctx):
                return StepResult(status=StepStatus.SKIPPED)
            result = await self.run(ctx)
            result.duration_ms = (time.perf_counter() - t0) * 1000
            return result
        except Exception as e:
            from kunlun.exceptions import PipelineStepError
            duration = (time.perf_counter() - t0) * 1000
            raise PipelineStepError(
                f"[{self.name}] 执行失败: {e}",
                detail={"step": self.name, "duration_ms": duration, "chapter": ctx.chapter},
            ) from e
```

#### Step 2：实现执行引擎 (`pipeline/engine.py`)

```python
"""管线执行引擎（对标 Haystack Pipeline + Prefect Flow）"""
import asyncio
from typing import Optional
from loguru import logger
from .base import PipelineContext, PipelineStep, StepResult, StepStatus
from .dag import topological_sort
from .checkpoint import CheckpointManager
from .middleware import MiddlewareChain


class Pipeline:
    """声明式管线定义"""
    
    def __init__(self, name: str, steps: list[PipelineStep]):
        self.name = name
        self.steps = steps
        self._step_map = {s.name: s for s in steps}
    
    def topological_order(self) -> list[PipelineStep]:
        """按依赖拓扑排序（对标 Haystack Pipeline DAG）"""
        return topological_sort(self.steps)


class PipelineEngine:
    """管线执行引擎"""
    
    def __init__(
        self,
        pipeline: Pipeline,
        checkpoint_dir: str = "",
        middleware: Optional[MiddlewareChain] = None,
    ):
        self.pipeline = pipeline
        self.checkpoint = CheckpointManager(checkpoint_dir)
        self.middleware = middleware or MiddlewareChain.default()
    
    async def run(self, ctx: PipelineContext) -> StepResult:
        """执行完整管线，支持检查点恢复"""
        logger.info(f"[{ctx.pipeline_id}] 管线启动: {self.pipeline.name}")
        
        # 从检查点恢复
        completed_steps = await self.checkpoint.load(ctx.pipeline_id)
        
        for step in self.pipeline.topological_order():
            if step.name in completed_steps:
                logger.debug(f"[{ctx.pipeline_id}] 跳过已完成步骤: {step.name}")
                continue
            
            # 中间件包装
            wrapped = self.middleware.wrap(step)
            
            try:
                result = await asyncio.wait_for(
                    wrapped(ctx),
                    timeout=step.timeout_seconds,
                )
            except asyncio.TimeoutError:
                result = StepResult(
                    status=StepStatus.FAILED,
                    error=f"步骤 {step.name} 超时 ({step.timeout_seconds}s)",
                )
            
            if not result.ok:
                if not step.optional:
                    logger.error(f"[{ctx.pipeline_id}] 关键步骤失败: {step.name}")
                    await self._compensate(ctx, up_to=step.name)
                    return result
                logger.warning(f"[{ctx.pipeline_id}] 可选步骤失败，继续: {step.name}")
            
            # 保存检查点
            await self.checkpoint.save(ctx.pipeline_id, step.name)
            
            # 让出事件循环
            await asyncio.sleep(0.001)
        
        await self.checkpoint.clear(ctx.pipeline_id)
        logger.info(f"[{ctx.pipeline_id}] 管线完成")
        return StepResult(status=StepStatus.COMPLETED)
    
    async def _compensate(self, ctx: PipelineContext, up_to: str):
        """Saga 补偿：反向执行已完成步骤的 rollback"""
        logger.warning(f"[{ctx.pipeline_id}] 执行补偿事务，范围: → {up_to}")
        for step in reversed(self.pipeline.topological_order()):
            if step.name == up_to:
                break
            try:
                await step.rollback(ctx)
            except Exception as e:
                logger.error(f"[{ctx.pipeline_id}] 补偿失败 [{step.name}]: {e}")
```

#### Step 3：实现中间件链 (`pipeline/middleware.py`)

```python
"""管线中间件链（对标 Django/ASP.NET Core 中间件洋葱模型）"""
import time
from typing import Callable, Awaitable
from functools import wraps
from loguru import logger
from kunlun.observability import record_generation_step
from .base import PipelineContext, StepResult, StepStatus


class MiddlewareChain:
    """中间件洋葱模型
    
    请求 → [计时] → [重试] → [熔断] → 步骤 → [熔断] → [重试] → [计时] → 响应
    """
    
    def __init__(self, middlewares: list[Callable]):
        self.middlewares = middlewares
    
    def wrap(self, step):
        """将中间件链包装到步骤上"""
        handler = step.__call__
        for mw in reversed(self.middlewares):
            handler = mw(handler)
        return handler
    
    @classmethod
    def default(cls):
        return cls([
            timing_middleware,
            retry_middleware,
            error_logging_middleware,
        ])


# ─── 内置中间件 ────────────────────────────

async def timing_middleware(call_next, step, ctx):
    """计时中间件：记录每步耗时到 observability"""
    t0 = time.perf_counter()
    result = await call_next(step, ctx)
    duration = time.perf_counter() - t0
    record_generation_step(step.name, duration)
    logger.debug(f"[Timing] {step.name}: {duration:.2f}s")
    return result


async def retry_middleware(call_next, step, ctx):
    """重试中间件：按步骤配置的重试次数重试"""
    result = await call_next(step, ctx)
    attempts = 1
    while not result.ok and attempts <= step.max_retries:
        wait = 2 ** (attempts - 1)  # 指数退避: 1s, 2s, 4s
        logger.warning(f"[Retry] {step.name} 第{attempts}次重试 (等待{wait}s)")
        await asyncio.sleep(wait)
        result = await call_next(step, ctx)
        attempts += 1
    return result


async def error_logging_middleware(call_next, step, ctx):
    """错误日志中间件：记录步骤级错误"""
    try:
        return await call_next(step, ctx)
    except Exception as e:
        logger.error(f"[{ctx.pipeline_id}] 步骤异常 [{step.name}]: {e}")
        raise
```

## 三、模块二：`kunlun/gacha/engine.py` — 滑动窗口熔断 + 多级降级

### 3.1 现状诊断

当前熔断实现：简单的60秒冷却，所有Key共享一个冷却时间，无半开探测。

### 3.2 目标：滑动窗口熔断器

对标 Resilience4j CircuitBreaker：
- **CLOSED** → 失败率 < 阈值，正常调用
- **OPEN** → 失败率 >= 阈值，直接拒绝（快速失败）
- **HALF_OPEN** → 冷却后尝试少量请求探测恢复
- 基于**滑动窗口**（最近N次调用）而非固定时间窗口

### 3.3 实施代码

```python
# kunlun/gacha/circuit_breaker.py（新建）

import time
from dataclasses import dataclass, field
from enum import Enum
from collections import deque
from loguru import logger


class CircuitState(str, Enum):
    CLOSED = "closed"          # 正常
    OPEN = "open"              # 熔断
    HALF_OPEN = "half_open"    # 探测


@dataclass
class CircuitBreakerConfig:
    """熔断器配置（对标 Resilience4j）"""
    failure_rate_threshold: float = 0.5      # 失败率阈值 50%
    sliding_window_size: int = 20            # 滑动窗口大小（最近N次）
    minimum_number_of_calls: int = 10        # 最小调用数才计算失败率
    wait_duration_in_open_state: float = 60  # OPEN → HALF_OPEN 等待时间
    permitted_calls_in_half_open: int = 2    # 半开状态允许的探测调用
    record_failure_predicate: callable = lambda e: True


class SlidingWindowCircuitBreaker:
    """滑动窗口熔断器（对标 Resilience4j CircuitBreaker）
    
    状态转换:
        CLOSED ──[失败率 >= 阈值]──→ OPEN
        OPEN   ──[等待超时]───────→ HALF_OPEN
        HALF_OPEN ──[成功]───────→ CLOSED
        HALF_OPEN ──[失败]───────→ OPEN
    """
    
    def __init__(self, name: str, config: CircuitBreakerConfig = None):
        self.name = name
        self.config = config or CircuitBreakerConfig()
        self.state = CircuitState.CLOSED
        self._window = deque(maxlen=self.config.sliding_window_size)
        self._half_open_calls = 0
        self._last_failure_time = 0.0
        self._state_changed_at = time.time()
    
    @property
    def failure_rate(self) -> float:
        if len(self._window) < self.config.minimum_number_of_calls:
            return 0.0
        failures = sum(1 for success, _ in self._window if not success)
        return failures / len(self._window)
    
    def _transition_to(self, new_state: CircuitState):
        if self.state != new_state:
            logger.info(
                f"[CircuitBreaker:{self.name}] {self.state.value} → {new_state.value} "
                f"(failure_rate={self.failure_rate:.2%}, window={len(self._window)})"
            )
            self.state = new_state
            self._state_changed_at = time.time()
    
    def acquire_permission(self) -> bool:
        """获取调用许可"""
        if self.state == CircuitState.CLOSED:
            return True
        
        if self.state == CircuitState.OPEN:
            elapsed = time.time() - self._state_changed_at
            if elapsed >= self.config.wait_duration_in_open_state:
                self._transition_to(CircuitState.HALF_OPEN)
                self._half_open_calls = 0
                return True
            return False
        
        if self.state == CircuitState.HALF_OPEN:
            if self._half_open_calls < self.config.permitted_calls_in_half_open:
                self._half_open_calls += 1
                return True
            return False
    
    def on_success(self):
        self._window.append((True, time.time()))
        if self.state == CircuitState.HALF_OPEN:
            self._transition_to(CircuitState.CLOSED)
    
    def on_failure(self, error: Exception = None):
        self._window.append((False, time.time()))
        self._last_failure_time = time.time()
        
        if self.config.record_failure_predicate(error):
            if self.state == CircuitState.HALF_OPEN:
                self._transition_to(CircuitState.OPEN)
            elif self.state == CircuitState.CLOSED:
                if (len(self._window) >= self.config.minimum_number_of_calls 
                    and self.failure_rate >= self.config.failure_rate_threshold):
                    self._transition_to(CircuitState.OPEN)
    
    def stats(self) -> dict:
        return {
            "name": self.name,
            "state": self.state.value,
            "failure_rate": round(self.failure_rate, 3),
            "window_size": len(self._window),
            "last_failure": self._last_failure_time,
        }
```

### 3.4 多级降级策略

```python
# kunlun/gacha/degradation.py（新建）

"""多级降级策略（对标 Istio Traffic Management + Resilience4j Fallback）"""

from typing import Optional
from loguru import logger
from kunlun.exceptions import GachaKeyExhaustedError, GachaRateLimitError


class DegradationLevel:
    """降级层级
    
    Level 0: 主模型（deepseek-reasoner）
    Level 1: 廉价模型（deepseek-chat, 低成本）
    Level 2: 备用 Key（openai_api_keys_backup 中的Key）
    Level 3: 本地缓存（无LLM，返回上次成功草稿）
    """
    PRIMARY = 0
    CHEAP = 1
    BACKUP_KEY = 2
    CACHED = 3


class DegradationRouter:
    """多级降级路由器"""
    
    def __init__(self, gacha_engine):
        self.engine = gacha_engine
        self._last_successful_draft: Optional[str] = None
        self._last_book_chapter: tuple[str, int] = ("", 0)
    
    async def route_with_degradation(
        self, prompt: str, mode: str, book_id: str = "", chapter: int = 0
    ) -> dict:
        """多级降级调用
        
        依次尝试: 主模型 → 廉价模型 → 备用Key → 缓存
        """
        strategies = [
            (DegradationLevel.PRIMARY, self._call_primary),
            (DegradationLevel.CHEAP, self._call_cheap),
            (DegradationLevel.BACKUP_KEY, self._call_backup_key),
            (DegradationLevel.CACHED, self._call_cached),
        ]
        
        for level, strategy in strategies:
            try:
                result = await strategy(prompt, mode)
                if result.get("success"):
                    # 缓存成功结果
                    self._last_successful_draft = result.get("best_text", "")
                    self._last_book_chapter = (book_id, chapter)
                    return result
            except (GachaRateLimitError, GachaKeyExhaustedError) as e:
                logger.warning(f"[Degradation] Level {level} 失败 → 降级: {e}")
                continue
        
        raise GachaKeyExhaustedError("所有降级策略均已耗尽")
    
    async def _call_primary(self, prompt, mode):
        return await self.engine._generate_impl(prompt, mode, use_primary=True)
    
    async def _call_cheap(self, prompt, mode):
        logger.info("[Degradation] 降级到廉价模型")
        return await self.engine._generate_impl(prompt, "single_fix", use_cheap=True)
    
    async def _call_backup_key(self, prompt, mode):
        logger.info("[Degradation] 降级到备用Key")
        return await self.engine._generate_impl(prompt, mode, use_backup_key=True)
    
    async def _call_cached(self, prompt, mode):
        logger.warning("[Degradation] 最终降级：返回缓存")
        if self._last_successful_draft:
            return {"success": True, "best_text": self._last_successful_draft, "degraded": True}
        raise GachaKeyExhaustedError("无可用缓存")
```

```python
# kunlun/core/plugin_manager.py（新建）
"""插件管理器（对标 pluggy PluginManager）"""

import inspect
from typing import Callable, Any
from loguru import logger


class HookSpec:
    """钩子规范 — 定义插件可实现的接口"""
    
    def __init__(self, name: str, description: str = ""):
        self.name = name
        self.description = description
    
    def __call__(self, func: Callable) -> Callable:
        func._hook_name = self.name
        func._hook_description = self.description
        return func


class PluginManager:
    """插件管理器（对标 pluggy PluginManager）
    
    使用方式:
        pm = PluginManager()
        
        # 注册钩子规范
        @pm.hookspec
        def on_chapter_complete(book_id: str, chapter: int, draft: str): ...
        
        # 注册插件实现
        @pm.register("on_chapter_complete", priority=10)
        async def my_plugin(book_id, chapter, draft):
            print(f"Chapter {chapter} complete!")
        
        # 触发钩子
        await pm.call("on_chapter_complete", book_id="test", chapter=1, draft="...")
    """
    
    def __init__(self):
        self._hooks: dict[str, list[tuple[int, Callable]]] = {}
        self._specs: dict[str, HookSpec] = {}
    
    def hookspec(self, func: Callable) -> Callable:
        """注册钩子规范"""
        spec = HookSpec(func.__name__, func.__doc__ or "")
        self._specs[func.__name__] = spec
        self._hooks.setdefault(func.__name__, [])
        return func
    
    def register(self, hook_name: str, priority: int = 0):
        """装饰器：注册插件实现"""
        def decorator(func: Callable):
            self._hooks.setdefault(hook_name, [])
            self._hooks[hook_name].append((priority, func))
            self._hooks[hook_name].sort(key=lambda x: -x[0])  # 高优先级在前
            logger.debug(f"[Plugin] 注册: {hook_name} (priority={priority})")
            return func
        return decorator
    
    async def call(self, hook_name: str, **kwargs) -> list[Any]:
        """触发钩子，按优先级调用所有注册的实现"""
        if hook_name not in self._hooks:
            logger.warning(f"[Plugin] 未注册的钩子: {hook_name}")
            return []
        
        results = []
        for priority, handler in self._hooks[hook_name]:
            try:
                if inspect.iscoroutinefunction(handler):
                    result = await handler(**kwargs)
                else:
                    result = handler(**kwargs)
                results.append(result)
            except Exception as e:
                logger.error(f"[Plugin] 钩子异常 [{hook_name}:{handler.__name__}]: {e}")
        return results
    
    def list_hooks(self) -> dict:
        """列出所有已注册的钩子"""
        return {
            name: {
                "spec": self._specs.get(name, HookSpec(name)).description,
                "handlers": len(handlers),
            }
            for name, handlers in self._hooks.items()
        }


# 全局插件管理器
plugin_manager = PluginManager()
```
