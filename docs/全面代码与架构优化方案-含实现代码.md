### P0-1: 消除 400+ 处 print 语句 → loguru

**现状**：项目存在 400+ 处 `print()` 语句（来自审计报告），与已有的 `loguru` 日志体系并存，污染生产控制台输出。

**方案对比**：

| 方案 | 优点 | 缺点 | 推荐 |
|------|------|------|:---:|
| 全局替换 `print` → `logger.info` | 简单直接 | 需区分调试/信息/错误级别 | ✅ |
| monkey-patch `builtins.print` | 零重构 | 无法区分日志级别，破坏第三方库 | ❌ |
| 仅替换生产路径 | 改动最小 | 开发调试失去便利性 | ❌ |

**推荐实现**：

```python
# kunlun/common/logging_patch.py
"""一次性消除 print 语句，通过预提交钩子阻止回归"""
import builtins
import sys
import logging
from loguru import logger

# 方案A：在 pre-commit 钩子中禁止 print（推荐）
# .pre-commit-config.yaml 已有配置，补充规则：
#   - repo: https://github.com/pre-commit/pre-commit-hooks
#     rev: v4.5.0
#     hooks:
#       - id: debug-statements  # 检测 breakpoint() 和 print()

# 方案B：运行时重定向（过渡期使用，生产环境启用）
def _patched_print(*args, **kwargs):
    """将 print 重定向到 loguru，仅在非开发环境生效"""
    from kunlun.config import settings
    if settings.is_production():
        logger.debug(" ".join(str(a) for a in args))
    else:
        builtins._original_print(*args, **kwargs)

# 在 main.py 启动时调用
def apply_print_patch():
    if not hasattr(builtins, '_original_print'):
        builtins._original_print = builtins.print
        builtins.print = _patched_print
```

# kunlun/api/error_handlers.py
"""统一异常处理层：类型化异常 → HTTP 状态码映射"""
from fastapi import Request
from fastapi.responses import JSONResponse
from kunlun.exceptions import (
    KunlunError, WriterError, PipelineError, GachaError,
    KGError, ExportError, AuditError, ConfigError, AuthError
)

# 异常 → HTTP 状态码映射表
EXCEPTION_STATUS_MAP = {
    AuthError: 401,
    ConfigError: 400,
    WriterError: 422,
    PipelineError: 500,
    GachaError: 502,      # 上游 LLM API 失败
    KGError: 500,
    ExportError: 422,
    AuditError: 422,
    KunlunError: 400,     # 兜底
}

async def kunlun_exception_handler(request: Request, exc: KunlunError):
    """统一 KunlunError 处理，保留 detail 字段"""
    status_code = EXCEPTION_STATUS_MAP.get(type(exc), 400)
    return JSONResponse(
        status_code=status_code,
        content={
            "success": False,
            "error": exc.message,
            "error_type": type(exc).__name__,
            "detail": exc.detail if hasattr(exc, 'detail') else None,
        }
    )

async def generic_exception_handler(request: Request, exc: Exception):
    """兜底处理：非预期异常记录 traceback，返回通用 500"""
    from loguru import logger
    logger.opt(exception=True).error("Unhandled exception")
    return JSONResponse(
        status_code=500,
        content={
            "success": False,
            "error": "Internal server error",
            "error_type": "InternalError",
        }
    )

# 在 main.py 中注册
def register_exception_handlers(app):
    for exc_cls in EXCEPTION_STATUS_MAP:
        app.add_exception_handler(exc_cls, kunlun_exception_handler)
    app.add_exception_handler(Exception, generic_exception_handler)
```

# kunlun/pipeline/steps.py
"""声明式管线步骤定义"""
from dataclasses import dataclass, field
from typing import Callable, Awaitable, Optional
from enum import StrEnum

class StepStatus(StrEnum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    SKIPPED = "skipped"

@dataclass
class PipelineStep:
    """单个管线步骤"""
    name: str
    handler: Callable[..., Awaitable[dict]]
    depends_on: list[str] = field(default_factory=list)
    retry_count: int = 3
    timeout_seconds: int = 30
    status: StepStatus = StepStatus.PENDING
    result: Optional[dict] = None

@dataclass
class PipelineCheckpoint:
    """检查点：支持断点续跑"""
    step_index: int
    step_name: str
    data: dict


# kunlun/pipeline/engine.py
"""声明式管线引擎"""
import asyncio
from loguru import logger
from typing import Any
from kunlun.exceptions import PipelineError

class PipelineEngine:
    """声明式管线执行引擎

    特性：
    - DAG 拓扑排序，自动处理步骤依赖
    - 步骤级重试（指数退避）
    - 检查点/断点续跑
    - 中间件链（before/after hooks）
    - 超时控制
    """

    def __init__(self, steps: list[PipelineStep], middlewares: list = None):
        # 拓扑排序：确保依赖步骤先执行
        self._sorted_steps = self._topological_sort(steps)
        self._middlewares = middlewares or []
        self._checkpoint: Optional[PipelineCheckpoint] = None

    def _topological_sort(self, steps: list[PipelineStep]):
        """DAG 拓扑排序"""
        step_map = {s.name: s for s in steps}
        in_degree = {s.name: len(s.depends_on) for s in steps}
        children = {s.name: [] for s in steps}

        for s in steps:
            for dep in s.depends_on:
                if dep in children:
                    children[dep].append(s.name)

        queue = [n for n, d in in_degree.items() if d == 0]
        result = []

        while queue:
            name = queue.pop(0)
            result.append(step_map[name])
            for child in children[name]:
                in_degree[child] -= 1
                if in_degree[child] == 0:
                    queue.append(child)

        if len(result) != len(steps):
            raise PipelineError("检测到循环依赖，管线步骤存在环形依赖")

        return result

    async def run(self, context: dict, resume_from: int = 0) -> dict:
        """执行管线

        Args:
            context: 共享上下文，步骤间通过 context 传递数据
            resume_from: 断点续跑起始索引

        Returns:
            最终的 context dict
        """
        start_idx = resume_from or 0

        for i, step in enumerate(self._sorted_steps[start_idx:], start=start_idx):
            if step.status == StepStatus.COMPLETED:
                logger.info(f"跳过已完成步骤: {step.name}")
                continue

            logger.info(f"执行步骤 [{i+1}/{len(self._sorted_steps)}]: {step.name}")

            # 中间件 before hook
            for mw in self._middlewares:
                context = await mw.before(step, context)

            # 步骤执行（带重试）
            for attempt in range(step.retry_count + 1):
                try:
                    step.status = StepStatus.RUNNING
                    result = await asyncio.wait_for(
                        step.handler(context),
                        timeout=step.timeout_seconds,
                    )
                    context.update(result)
                    step.result = result
                    step.status = StepStatus.COMPLETED
                    break
                except asyncio.TimeoutError:
                    logger.warning(f"步骤 {step.name} 超时 (尝试 {attempt+1}/{step.retry_count+1})")
                    if attempt == step.retry_count:
                        raise PipelineError(f"步骤 {step.name} 超时，重试 {step.retry_count} 次后仍失败")
                    await asyncio.sleep(2 ** attempt)  # 指数退避
                except Exception as e:
                    logger.error(f"步骤 {step.name} 失败: {e}")
                    if attempt == step.retry_count:
                        step.status = StepStatus.FAILED
                        # 保存检查点
                        self._checkpoint = PipelineCheckpoint(
                            step_index=i, step_name=step.name, data=context
                        )
                        raise PipelineError(f"步骤 {step.name} 最终失败: {e}") from e
                    await asyncio.sleep(2 ** attempt)

            # 中间件 after hook
            for mw in self._middlewares:
                context = await mw.after(step, context)

        return context

    @property
    def last_checkpoint(self) -> Optional[PipelineCheckpoint]:
        return self._checkpoint
```

### P0-4: 滑动窗口熔断器 + 多级降级策略

**现状**：`gacha/engine.py` 使用简单 60 秒冷却机制，无熔断降级。LLM API 不稳定时直接报错。

**当前架构**：
```
模型调用 → 失败 → sleep(60) → 重试 → 仍失败 → 抛异常
```

**推荐架构**：
```
模型调用 → 失败 → 滑动窗口判断
  ├── 间隔错误 → 重试（指数退避）
  ├── 连续 5 次 50% 失败率 → 熔断开启（30s）→ 降级到廉价模型
  ├── 廉价模型也失败 → 降级到备用 API Key
  └── 全部失败 → 降级到缓存/模拟
```

**实现代码**：

```python
# kunlun/gacha/circuit_breaker.py
"""滑动窗口熔断器 + 多级降级策略"""
import time
import asyncio
from collections import deque
from enum import StrEnum
from dataclasses import dataclass, field
from loguru import logger

class CircuitState(StrEnum):
    CLOSED = "closed"          # 正常
    OPEN = "open"              # 熔断中
    HALF_OPEN = "half_open"    # 半开（探测恢复）

@dataclass
class SlidingWindowStats:
    """滑动窗口统计"""
    window_size: int = 10
    threshold: float = 0.5
    failures: deque = field(default_factory=lambda: deque(maxlen=10))
    successes: deque = field(default_factory=lambda: deque(maxlen=10))

    @property
    def failure_rate(self) -> float:
        total = len(self.failures) + len(self.successes)
        if total == 0:
            return 0.0
        return len(self.failures) / total

    def record_success(self):
        now = time.time()
        self.successes.append(now)
        # 清理超过窗口的旧记录
        self._cleanup()

    def record_failure(self):
        now = time.time()
        self.failures.append(now)
        self._cleanup()

    def _cleanup(self):
        window_start = time.time() - self.window_size
        while self.failures and self.failures[0] < window_start:
            self.failures.popleft()
        while self.successes and self.successes[0] < window_start:
            self.successes.popleft()


class CircuitBreaker:
    """滑动窗口熔断器

    用法：
        breaker = CircuitBreaker(failure_threshold=0.5, recovery_timeout=30)
        async with breaker:
            result = await call_llm()
    """

    def __init__(self, failure_threshold: float = 0.5,
                 window_size: int = 10, recovery_timeout: int = 30):
        self._state = CircuitState.CLOSED
        self._stats = SlidingWindowStats(window_size=window_size)
        self._threshold = failure_threshold
        self._recovery_timeout = recovery_timeout
        self._opened_at: float = 0

    async def __aenter__(self):
        if self._state == CircuitState.OPEN:
            if time.time() - self._opened_at > self._recovery_timeout:
                self._state = CircuitState.HALF_OPEN
                logger.info("熔断器进入半开状态，尝试探测...")
            else:
                remaining = self._recovery_timeout - (time.time() - self._opened_at)
                raise CircuitBreakerOpenError(
                    f"熔断器开启中，{remaining:.0f}秒后重试"
                )
        return self

    async def __aexit__(self, exc_type, exc_val, exc_tb):
        if exc_type is None:
            self._on_success()
        elif issubclass(exc_type, Exception):
            self._on_failure()
        return False  # 不吞异常

    def _on_success(self):
        self._stats.record_success()
        if self._state == CircuitState.HALF_OPEN:
            self._state = CircuitState.CLOSED
            logger.info("熔断器恢复，进入关闭状态")

    def _on_failure(self):
        self._stats.record_failure()
        if self._state == CircuitState.HALF_OPEN:
            self._state = CircuitState.OPEN
            self._opened_at = time.time()
        elif (self._state == CircuitState.CLOSED
              and self._stats.failure_rate >= self._threshold):
            self._state = CircuitState.OPEN
            self._opened_at = time.time()
            logger.warning(
                f"熔断器开启！失败率 {self._stats.failure_rate:.1%} "
                f"超过阈值 {self._threshold:.1%}"
            )


class CircuitBreakerOpenError(Exception):
    pass


# kunlun/gacha/fallback.py
"""多级降级策略"""
from typing import Protocol, Any
from loguru import logger
from kunlun.gacha.circuit_breaker import CircuitBreaker

class LLMProvider(Protocol):
    async def generate(self, prompt: str, **kwargs) -> str: ...

class DegradationChain:
    """多级降级链：主模型 → 廉价模型 → 备用Key → 缓存 → 兜底"""

    def __init__(
        self,
        primary: LLMProvider,
        fallback_model: LLMProvider = None,
        fallback_keys: list[str] = None,
        cache: Any = None,
    ):
        self._primary = primary
        self._fallback_model = fallback_model
        self._fallback_keys = fallback_keys or []
        self._cache = cache
        self._breaker = CircuitBreaker(failure_threshold=0.5, recovery_timeout=30)

    async def generate(self, prompt: str, **kwargs) -> str:
        errors = []

        # Level 1: 主模型
        try:
            async with self._breaker:
                return await self._primary.generate(prompt, **kwargs)
        except Exception as e:
            errors.append(f"主模型: {e}")
            logger.warning(f"主模型失败，降级中... ({e})")

        # Level 2: 廉价回退模型
        if self._fallback_model:
            try:
                logger.info("降级到廉价模型")
                return await self._fallback_model.generate(prompt, **kwargs)
            except Exception as e:
                errors.append(f"廉价模型: {e}")

        # Level 3: 备用 API Key 轮换
        for idx, key in enumerate(self._fallback_keys):
            try:
                logger.info(f"降级到备用 Key {idx+1}/{len(self._fallback_keys)}")
                return await self._primary.generate(prompt, api_key=key, **kwargs)
            except Exception as e:
                errors.append(f"备用Key{idx+1}: {e}")

        # Level 4: 缓存命中
        if self._cache:
            cached = await self._cache.get(prompt)
            if cached:
                logger.info("降级到缓存")
                return cached

        # Level 5: 完全失败
        raise DegradationExhaustedError(
            f"所有降级策略均失败: {' → '.join(errors)}"
        )


class DegradationExhaustedError(Exception):
    pass
```

### P1-3: 添加全局超时控制

**现状**：`api/index.ts` 的 `fetch` 无 `AbortController`，网络慢时无限挂起。

```typescript
// frontend/src/api/index.ts — 添加超时控制
const DEFAULT_TIMEOUT_MS = 30_000  // 30秒默认超时

async function request<T>(
  method: string,
  path: string,
  body?: Record<string, unknown>,
  queryParams?: Record<string, any>,
  timeoutMs: number = DEFAULT_TIMEOUT_MS,
): Promise<T> {
  const controller = new AbortController()
  const timer = setTimeout(() => controller.abort(), timeoutMs)

  try {
    let url = `${BASE}${path}`
    // ... queryParams 处理同前

    const res = await fetch(url, {
      method,
      headers: body ? { 'Content-Type': 'application/json' } : {},
      body: body ? JSON.stringify(body) : undefined,
      signal: controller.signal,  // 关键：绑定 AbortController
    })

    const data = await res.json()
    if (!res.ok) throw new ApiError(res.status, data.detail || data.error || `请求失败: ${res.status}`)
    return data as T
  } catch (e) {
    if (e instanceof DOMException && e.name === 'AbortError') {
      throw new ApiError(408, `请求超时 (${timeoutMs / 1000}s)`)
    }
    throw e
  } finally {
    clearTimeout(timer)
  }
}
```

import pluggy

hookspec = pluggy.HookspecMarker("kunlun")
hookimpl = pluggy.HookimplMarker("kunlun")

class KunlunPluginSpec:
    """昆仑插件钩子规范"""
    @hookspec
    def on_pipeline_step_start(self, step_name: str, context: dict): ...

    @hookspec
    def on_pipeline_step_complete(self, step_name: str, result: dict): ...

    @hookspec
    def on_chapter_generated(self, chapter: int, draft: str): ...

class PluginManager:
    def __init__(self):
        self._pm = pluggy.PluginManager("kunlun")
        self._pm.add_hookspecs(KunlunPluginSpec)

    def register(self, plugin):
        self._pm.register(plugin)

    def discover_entry_points(self):
        """自动发现 pip 安装的插件"""
        import importlib.metadata
        for ep in importlib.metadata.entry_points(group="kunlun.plugins"):
            plugin_class = ep.load()
            self.register(plugin_class())

## 九、注意事项

1. **渐进式改造**：所有重构采用"新增 + 迁移"策略，保留旧代码直到新代码验证通过
2. **测试先行**：每项改造前先为现有功能补充测试（当前测试断言过于宽松）
3. **功能标记**：大规模改造（管线、路由）使用功能开关，支持灰度发布
4. **数据库迁移**：仓储模式改造涉及数据访问层变更，需配合 Alembic 迁移脚本
5. **回滚方案**：每条改造分支独立，单条出问题只回滚该分支，不影响其他改造
